
const heading = document.querySelector("#basket-heading");

const fruits = document.querySelectorAll(".fruit");

fruits.forEach((fruit, index) => {
    if ((index + 1) % 2 === 0) {
        fruit.style.backgroundColor = "brown";
        fruit.style.color = "white";
    }
});
